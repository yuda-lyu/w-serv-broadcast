import WConverhpServer from 'w-converhp/src/WConverhpServer.mjs'
import WServBroadcastServer from './src/WServBroadcastServer.mjs'

let opt = {
    port: 8080,
    apiName: 'api',
    pathStaticFiles: '.', //要存取專案資料夾下web.html, 故不能給dist
    verifyConn: () => {
        return true
    },
}

//instWConverServer
let instWConverServer = new WConverhpServer(opt)

//instWConverServer
instWConverServer = new WServBroadcastServer(instWConverServer)

let n = 0
setInterval(() => {
    n++
    instWConverServer.broadcast(`n=${n}`)
}, 1500)

instWConverServer.on('clientEnter', function(clientId, data) {
    console.log(`Server[port:${opt.port}]: clientEnter`, clientId, data)
})
instWConverServer.on('clientLeave', function(clientId, data) {
    console.log(`Server[port:${opt.port}]: clientLeave`, clientId, data)
})
instWConverServer.on('clientChange', function(numClients) {
    console.log(`Server[port:${opt.port}]: clientChange`, numClients)
})
instWConverServer.on('error', function(err) {
    console.log(`Server[port:${opt.port}]: error`, err)
})
instWConverServer.on('handler', function(data) {
    // console.log(`Server[port:${opt.port}]: handler`, data)
})

instWConverServer Object

輸入通訊服務實體物件，須為具emit、on、off函數之eventemitter3事件物件(例如wsemi之evem所建立者)，可使用例如WConverhpServer等建立；有物件而缺emit、on或off時於建構時拋錯，且不留下任何計時器與監聽；未給或非有效物件時改以evem建立不連線之事件物件(並印出提示)，可由emit('execute', func, input, pm)手動驅動。同一事件物件只能建立一個本伺服器，重建前須先呼叫clearBroadcast

Returns:

回傳事件物件，可監聽clientEnter、clientLeave、clientChange、error事件，並保留instWConverServer原有之execute、upload、download、handler事件；另掛載broadcast、getClients、getMessages、clearBroadcast函數。本套件只處理函數名為[sys:polling]之execute，其他函數名須由應用端自行註冊execute監聽並回應，否則該請求懸置至前端逾時；本套件之execute監聽須先於「對所有函數名皆回應」之監聽註冊，否則輪詢由該監聽回應。輪詢之clientId須為非空字串，任意字串(含__proto__、constructor、含.或[]者)皆為一般識別，回覆一律為陣列；clientId不是非空字串者拒絕該次輪詢([sys:polling] invalid clientId)，不發事件、不建立在線身分。clientEnter與clientLeave帶(clientId, data)，clientEnter之data為該客戶端首次輪詢時所送之輸入物件，clientLeave之data為其最後一次輪詢時所送者；clientChange帶當前存活客戶端數量(含剛輪詢而尚未發出clientEnter者)。broadcast以呼叫當下之值(複製)廣播給當前在線之客戶端，data須可序列化(同通訊服務之規則)，否則同批廣播於輪詢時遺失。在線逾時固定10秒，傳遞為至多一次：客戶端兩次成功輪詢之間隔達10秒時視為離線並丟棄其未取走之廣播，回覆送不到(例如不可序列化之data、傳輸失敗)時同批之廣播遺失。clearBroadcast為停止並釋放：清除本套件之計時器與在線偵測(行程可立即結束)、移除所掛之execute監聽(其後不再回應輪詢，含於同一次execute派發中先註冊之監聽內停止者；停止後之輪詢由同一事件物件上其他execute監聽回應，無其他監聽時由通訊服務立即拒絕，另有不回應[sys:polling]之監聽時懸置至前端逾時)，並對已送達clientEnter而尚未送達clientLeave之客戶端補發clientLeave與clientChange使進出成對；停止前已排定而尚未送達之事件不再派發，補發之後不再派發本套件之事件，惟呼叫當下那次派發之其餘監聽器仍會收到；停止後之監聽器拋錯(含補發事件之監聽器)以console.error輸出；停止後broadcast無作用、getClients回傳[]、getMessages回傳{}，可重複呼叫。getMessages回傳以clientId為鍵之一般物件(複製)。error事件之內容一律為{ msg, err }，msg為描述字串、err為原因(原始錯誤或出錯之值)：輪詢處理失敗為{ msg: '[sys:polling] error', err }；本套件之error於派發當下無error監聽者時改以console.error輸出；監聽器之同步拋錯不會使行程崩潰，改以error事件通知，內容為{ msg: 'listener of event[x] error', err }(x為事件名、err為所拋之值)，無error監聽者時改以console.error輸出；error監聽器自身拋錯亦以console.error輸出。惟依EventEmitter規範，同一次派發中先拋錯之監聽器會中止該次派發，其後之監聽器不再被呼叫；async監聽器之reject不被攔截，須由監聽器自行處理
